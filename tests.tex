% Nome formatado para a Ficha Catalográfica: SOBRENOME, PRENOMES
\newcommand{\autorreferenciacatalografica}{%
  \ifx\ultimosobrenomeautorstr\@empty
    \begingroup
      \def\\{, }%
      \MakeTextUppercase{\@author}%
    \endgroup
  \else
    \MakeTextUppercase{\ultimosobrenomeautorstr, \prenomesautorstr}%
  \fi
}

% Extração recursiva de iniciais dos prenomes (separados por espaço)
\def\ft@process@initials#1 {%
  \def\ft@temp{#1}%
  \def\ft@stop{ENDINITIALS}%
  \ifx\ft@temp\ft@stop
    % Sentinela encontrado, encerra a recursão
  \else
    \ft@check@particle{#1}%
    \expandafter\ft@process@initials
  \fi
}

% Invoca a macro adicionando o sentinela de parada
\def\ft@invoke@initials#1{\ft@process@initials#1 ENDINITIALS \@empty}

\newcommand{\iniciais}[1]{%
  \expandafter\ft@invoke@initials\expandafter{#1}%
}

% Filtro de partículas (inclui as notórias estrangeiras)
\def\ft@check@particle#1{%
  \def\ft@temp{#1}%
  \ifx\ft@temp\@empty\else % Proteção contra espaços duplos
    \ifthenelse{%
      \equal{#1}{da}\OR\equal{#1}{de}\OR\equal{#1}{do}\OR%
      \equal{#1}{das}\OR\equal{#1}{dos}\OR\equal{#1}{e}\OR%
      \equal{#1}{Da}\OR\equal{#1}{De}\OR\equal{#1}{Do}\OR%
      \equal{#1}{Das}\OR\equal{#1}{Dos}\OR\equal{#1}{E}\OR%
      \equal{#1}{van}\OR\equal{#1}{von}\OR\equal{#1}{la}\OR\equal{#1}{di}\OR%
      \equal{#1}{del}\OR\equal{#1}{du}\OR\equal{#1}{des}\OR\equal{#1}{der}\OR%
      \equal{#1}{ten}\OR\equal{#1}{zu}\OR\equal{#1}{le}\OR\equal{#1}{y}%
    }{%
      % Ignora a partícula e não imprime inicial
    }{%
      \MakeTextUppercase{\expandafter\@car#1\@nil}.\space
    }%
  \fi
}

% Iniciais dos prenomes
\newcommand{\iniciaisautor}{%
  \ifx\prenomesautorstr\@empty
    \MakeTextUppercase{\expandafter\@car\@author\@nil}.%
  \else
    \iniciais{\prenomesautorstr}%
  \fi
}

% Nome para a Referência Bibliográfica: SOBRENOME, I. N.
\newcommand{\autorreferenciabibliografica}{%
  \ifx\ultimosobrenomeautorstr\@empty
    \begingroup
      \def\\{, }%
      \MakeTextUppercase{\@author}%
    \endgroup
  \else
    \MakeTextUppercase{\ultimosobrenomeautorstr}, \iniciaisautor
  \fi
}
